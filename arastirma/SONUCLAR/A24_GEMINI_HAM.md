import glob
import yaml
import pandas as pd
from pathlib import Path

# Ham raporların bulunduğu dizin ve hedef tablo dosyası
REPORTS_DIR = "./movies"
OUTPUT_MATRIX_CSV = "./master_index_matrix.csv"

def parse_markdown_frontmatter(filepath):
    """
    Markdown dosyasını açar, en üstte '---' işaretleri arasına alınmış
    YAML meta-verisini sözlük (dictionary) olarak döndürür.
    """
    with open(filepath, 'r', encoding='utf-8') as file:
        content = file.read()
        
    # Dosya başının YAML formatına uygun olup olmadığını kontrol et
    if not content.startswith('---'):
        return None
        
    # Metni '---' karakterlerine göre böl ve index 1'i (YAML içeriği) al
    try:
        yaml_block = content.split('---')[1]
        return yaml.safe_load(yaml_block)
    except (IndexError, yaml.YAMLError) as e:
        print(f"Hata! {filepath} dosyasındaki YAML yapısı bozuk: {e}")
        return None

def generate_master_matrix():
    """
    Tüm markdown dosyalarını tarar, geçerli olanları listeler ve 
    film bazında bir puan (score) matrisi DataFrame'ine dönüştürür.
    """
    valid_records = []
    
    # İlgili klasördeki tüm markdown (.md) dosyalarını bul
    md_files = glob.glob(f"{REPORTS_DIR}/**/*.md", recursive=True)
    
    for file_path in md_files:
        metadata = parse_markdown_frontmatter(file_path)
        
        if not metadata:
            continue
            
        # Uzlaştırma: Sadece geçerli ve bağlayıcı ("complete") olan raporları özet kayda al.
        # "superseded", "non-binding", "invalid" olan raporları hesaplamaya KAtMA.
        if metadata.get('status') == 'complete':
            record = {
                'Movie_ID': metadata.get('movie_id'),
                'Rubric': metadata.get('rubric_id'),
                'Score': metadata.get('score'),
                'Timestamp': metadata.get('timestamp')
            }
            valid_records.append(record)
            
    if not valid_records:
        print("Uyarı: Matrise eklenecek geçerli (complete) bir rapor bulunamadı.")
        return
        
    df = pd.DataFrame(valid_records)
    
    # 8. Kural (Uzlaştırma): Aynı film/rubrik ikilisi için iki tane complete varsa 
    # bu bir veri çakışmasıdır, ancak pivot işlemi sırasında hata vermemesi için uyarı.
    if df.duplicated(subset=['Movie_ID', 'Rubric']).any():
        print("KRİTİK UYARI: Aynı film için aynı rubrikte birden fazla 'complete' statülü dosya tespit edildi! (Uzlaştırma Kuralı 8 İhlali)")
    
    # Film ID'si satırlarda, Rubrik ID'si sütunlarda olacak şekilde Pivot Matrisi yarat.
    # Çakışma durumunda (eğer engellenememişse) ortalama (mean) alır.
    matrix_df = df.pivot_table(
        index='Movie_ID', 
        columns='Rubric', 
        values='Score', 
        aggfunc='mean'
    )
    
    # CSV formatında diske (otomatik üretilmiş salt okunur dosya olarak) kaydet.
    matrix_df.to_csv(OUTPUT_MATRIX_CSV)
    print(f"Başarılı: Ana dizin matrisi yeniden oluşturuldu -> {OUTPUT_MATRIX_CSV}")

if __name__ == "__main__":
    generate_master_matrix()
